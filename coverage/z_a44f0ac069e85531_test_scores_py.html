13def votes_of(rows: list[dict[str, object]]) -> pd.DataFrame: 

14 """Returns a small table of votes from the rows that differ.""" 

15 defaults = { 

16 'judge': 'Judge', 'judge_party': -1.0, 'court_num': 1, 

17 'type_criminal': False, 'review_standard': 'deferential', 

18 'vote_reversal': False, 'judge_dissented': False, 

19 'judge_concurred': False, 'judge_author': True, 

20 'district_party': np.nan, 'alignment': np.nan} 

21 table = pd.DataFrame([ 

22 {**defaults, 'case_id': f'c{i}', 'judge_nid': 1, **row} 

23 for i, row in enumerate(rows)]) 

24 table.index = pd.Index([f'v{i}' for i in range(len(table))]) 

25 return table

28def cast(standard: str, total: int, reversals: int, **row: object) -> list[dict[str, object]]: 

29 """Returns the votes of one judge in one cell.""" 

30 return [ 

31 {'review_standard': standard, 'vote_reversal': i < reversals, **row} 

32 for i in range(total)]

35def test_activism_formula() -> None: 

36 # Judge Wood's example in Flexing Judicial Muscle: 4 reversals of 53 under 

37 # deferential standards and 13 of 49 under de novo review. 

38 table = votes_of( 

39 cast('deferential', 53, 4) + cast('non_deferential', 49, 13)) 

40 scored = scores.score_judges( 

41 table, method = 'original', minimum = 0, cell_minimum = 1, 

42 circuit = 'none') 

43 assert scored['activism_raw'][0] == pytest.approx(4 / 53 - 13 / 49) 

44 assert scored['activism_super_panel'][0] == 0 

45 assert scored['activism_adjusted'][0] == pytest.approx(4 / 53 - 13 / 49)

48def test_partisanship_formula() -> None: 

49 # A Democratic appointee who defers more to Democratic district judges 

50 # under deferential standards is partisan, and a Republican appointee 

51 # who does the same is not. 

52 rows = ( 

53 cast('deferential', 4, 0, district_party = -1.0) 

54 + cast('deferential', 4, 2, district_party = 1.0) 

55 + cast('non_deferential', 4, 1, district_party = -1.0) 

56 + cast('non_deferential', 4, 1, district_party = 1.0)) 

57 democrat = scores.score_judges( 

58 votes_of(rows), minimum = 0, cell_minimum = 1, circuit = 'none') 

59 assert democrat['partisanship_raw'][0] == pytest.approx(0.5) 

60 # The rates of all judges are the same for both parties, so the 

61 # super-panel takes nothing away. 

62 assert democrat['partisanship_super_panel'][0] == pytest.approx(0) 

63 republican = scores.score_judges( 

64 votes_of([{**row, 'judge_party': 1.0} for row in rows]), 

65 minimum = 0, cell_minimum = 1, circuit = 'none') 

66 assert republican['partisanship_raw'][0] == pytest.approx(-0.5)

84def test_scores_find_traits(simulation: Simulation) -> None: 

85 table = scores.score_judges(simulation.votes) 

86 merged = table.set_index('judge_nid').join( 

87 simulation.traits, rsuffix = '_true') 

88 assert merged['qualified'].all() 

89 expected = { 

90 'ideology': 0.6, 'activism': 0.45, 'partisanship': 0.3, 

91 'independence': 0.7}

93 correlation = merged[score].corr( 

94 merged[f'{score}_true'], method = 'spearman')

98def test_scales(simulation: Simulation) -> None: 

99 table = scores.score_judges(simulation.votes) 

100 for score in ('activism', 'partisanship', 'independence'): 

101 assert table[score].min() == pytest.approx(0) 

102 assert table[score].max() == pytest.approx(100) 

103 assert table['ideology'].abs().max() == pytest.approx(100) 

104 # The table is sorted from the most liberal judge, and liberal judges 

105 # are mostly Democratic appointees. 

106 assert table['ideology'].is_monotonic_increasing 

107 liberal = table['ideology'] < 0 

108 assert (table.loc[liberal, 'judge_party'] == -1).mean() > 0.7 

109 assert {'circuit', 'votes', 'interactions', 'traveling_votes'} <= set( 

110 table.columns) 

111 assert table['traveling_votes'].sum() > 0

114def test_methods(simulation: Simulation) -> None: 

115 typology = scores.score_judges(simulation.votes) 

116 original = scores.score_judges(simulation.votes, method = 'original') 

117 # The original Ideology Score puts its zero at the judge whose rank is 

118 # the number of Democratic appointees. 

119 democrats = int((original['judge_party'] == -1).sum()) 

120 ranked = np.sort(original['ideology'].to_numpy()) 

121 assert ranked[democrats - 1] <= 0 <= ranked[democrats] 

122 assert (original['activism_super_panel'] == 0).all() 

123 assert (typology['activism_super_panel'] != 0).any() 

124 # The per-judge circuit adjustment is the same for every judge of a 

125 # circuit. 

126 per_judge = original.groupby('circuit')['ideology_circuit'].nunique() 

127 assert (per_judge == 1).all() 

128 none = scores.score_judges(simulation.votes, circuit = 'none') 

129 assert (none['ideology_circuit'] == 0).all() 

130 full = scores.score_judges(simulation.votes, circuit = 'full') 

131 assert (full['ideology_circuit'] != 0).any()
