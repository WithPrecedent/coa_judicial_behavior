10 

11import coa_judicial_behavior

13STUDIES = pathlib.Path(__file__).parents[1] / 'studies' 

14 

15 

16def test_judicial_behavior_study( 

17 roster: courtpy.judges.Roster, 

18 tmp_path: pathlib.Path) -> None: 

19 # Judge Kay of the First Circuit reverses as often under deferential 

20 # standards as under de novo review. 

21 write_cases(tmp_path / 'data' / 'court_listener', 600, activist = 'Kay') 

22 idea = chrisjen.Idea.create(STUDIES / 'judicial_behavior.toml') 

23 loader = idea['load_court_listener_parameters'] 

24 loader['download'] = 'none' 

25 loader['workers'] = 1 

26 loader['rulebooks'] = ( 

27 f'federal, {coa_judicial_behavior.RULEBOOK.as_posix()}') 

28 idea['score_judges_parameters']['train'] = 'year <= 2015' 

29 idea['score_judges_parameters']['minimum'] = 100 

30 idea['score_judges_parameters']['clusters'] = 3 

31 idea['random_forest_parameters']['n_estimators'] = 20 

32 # Eleven fictional judges cannot tell apart as many attributes of 

33 # judges as the study uses, which would make the probit model singular, 

34 # so only a few are kept. 

35 idea['keep_columns_parameters']['columns'] = [ 

36 'judge_ideology', 'judge_activism', 'judge_partisanship', 

37 'judge_independence', 'colleagues_ideology', 'partisan_pressure', 

38 'activist_pressure', 'ideological_pressure', 'expected_reversal', 

39 'type_criminal', 'review_deferential', 'review_de_novo', 

40 'district_party', 'court_num', 'year'] 

41 idea['group_k_fold_parameters']['n_splits'] = 3 

42 project = coa_judicial_behavior.Project.create(idea, clerk = tmp_path) 

43 result = project.result 

44 assert result.label == 'vote_reversal' 

45 techniques = [entry['technique'] for entry in result.history] 

46 assert techniques[:8] == [ 

47 'load_court_listener', 'code_parties', 'code_case_type', 

48 'code_outcome', 'code_judges', 'code_politics', 

49 'code_district_judge', 'code_review'] 

50 assert {'review_votes', 'score_judges', 'validate_scores'} <= set( 

51 techniques) 

52 assert (tmp_path / 'data' / 'appeals.parquet').is_file() 

53 # Every fictional appeal has three judges in the roster, so each has 

54 # three votes. 

55 assert len(result.data) == 1800 

56 tables = result.tables 

57 scored = tables['score_judges'].set_index('judge') 

58 assert len(scored) == 11 

59 assert scored['qualified'].all() 

60 # Every district judge was found, so every vote has its super-panel. 

61 assert scored['district_interactions'].sum() == scored['votes'].sum() 

62 # The judge who does not defer is the most activist of the First 

63 # Circuit. 

64 first = scored[scored['circuit'] == 1] 

65 assert first['activism'].idxmax().startswith('Kay') 

66 # Republican appointees reverse Democratic district judges, so they are 

67 # conservative. 

68 assert ( 

69 scored.loc[scored['judge_party'] == 1, 'ideology'].mean() 

70 > scored.loc[scored['judge_party'] == -1, 'ideology'].mean()) 

71 assert len(tables['typology']) == 3 

72 tests = tables['validate_scores'] 

73 assert {'dissent', 'partisan_reversal', 'civil_rights'} <= set( 

74 tests['test']) 

75 assert { 

76 'summarize', 'fixest_coefficients', 'analyst_comparison', 

77 'scorecard', 'permutation_importance'} <= set(tables) 

78 assert 'judge_activism' in tables['fixest_coefficients'].index 

79 assert 'roc_curve' in result.figures 

80 assert 0 <= result.metrics['roc_auc'] <= 1 

81 folder = project.export() 

82 assert (folder / 'tables' / 'score_judges.csv').is_file()
