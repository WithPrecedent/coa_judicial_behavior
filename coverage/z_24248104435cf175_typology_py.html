1"""Sorts judges into judging styles by their four scores. 

2 

3*A Typology of Judging Styles*, 107 Nw. U. L. Rev. 1757 (2013), grouped the 

4178 judges with at least 200 interactions by Ward's cluster analysis of four 

5scores, each from 0 to 100: the Activism, Independence, and Partisanship 

6Scores and the absolute value of the Ideology Score (so that a judge at -60 

7and one at 60 are alike in how ideological they are). A group counted as a 

8style only if its mean differed from every other group's by at least one 

9standard deviation in at least one score, which left nine styles. The 

10article named them from their mean scores: 

11 

12| Style | Ideological | Activism | Partisanship | Independence | 

13| --- | --- | --- | --- | --- | 

14| Trailblazing | High | Low | Low | High | 

15| Consensus Building | High | Below average | Below average | Lowest | 

16| Stalwart | Highest | Highest | Highest | Below average | 

17| Regulating | Average | Above average | High | Above average | 

18| Steadfast | Average | Elevated | Average | Highest | 

19| Collegial | Below average | Below average | Below average | Below average | 

20| Incrementalist | Slightly below | Slightly below | Elevated | Below average | 

21| Minimalist | Average | Lowest | Average | Slightly above | 

22| Error Correcting | Slightly below | High | Slightly below | Average | 

23 

24`classify` repeats the analysis (with scipy's Ward linkage, where the article 

25used Stata's) and names each group for the style whose profile is closest to 

26the group's standardized mean scores, matching each style to one group. The 

27names are a guide to reading the groups, not a finding: the groups of other 

28judges and years need not resemble the article's, and the table of profiles 

29shows how far each group is from its style. 

30 

31Contents: 

32 PROTOTYPES: the profile of each style, in standard deviations from the 

33 mean of each score. 

34 STYLES: the names of the nine styles, in the order of the article. 

35 Typology: the styles of the judges. 

36 classify: sorts judges into styles by Ward's cluster analysis. 

37 

38"""

110def classify( 

111 scores: pd.DataFrame, 

112 clusters: int = 9, 

113 *, 

114 names: Sequence[str] | None = None, 

115 only_qualified: bool = True) -> Typology: 

116 """Sorts judges into styles by Ward's cluster analysis. 

117 

118 Args: 

119 scores: the table of scores (see `score_judges`), with "ideology", 

120 "activism", "partisanship", and "independence" (and "qualified", 

121 if `only_qualified`). 

122 clusters: the number of clusters. Defaults to 9, as in the article. 

123 names: the names of the styles to match the clusters to. Defaults to 

124 `None`, in which case they are `STYLES` if `clusters` is 9, and 

125 else the clusters are named by their numbers. 

126 only_qualified: whether to classify only the judges who define the 

127 scales of the scores. Defaults to `True`. 

128 

129 Raises: 

130 ValueError: if there are fewer judges to classify than clusters. 

131 

132 Returns: 

133 The styles of the judges. 

134 

135 """ 

136 features = pd.DataFrame({ 

137 'ideological': scores['ideology'].abs(), 

138 'activism': scores['activism'], 

139 'partisanship': scores['partisanship'], 

140 'independence': scores['independence']}, index = scores.index) 

141 keep = features.notna().all(axis = 1) 

142 if only_qualified and 'qualified' in scores.columns: 142 ↛ 144line 142 didn't jump to line 144 because the condition on line 142 was always true

143 keep &= scores['qualified'].astype(bool) 

144 features = features[keep] 

145 if len(features) < max(clusters, 2): 

146 message = ( 

147 f'{len(features)} judges cannot be sorted into {clusters} ' 

148 f'clusters') 

149 raise ValueError(message) 

150 tree = hierarchy.linkage(features.to_numpy(dtype = 'float64'), 'ward') 

151 found = hierarchy.fcluster(tree, t = clusters, criterion = 'maxclust') 

152 # The clusters are numbered in the order that their judges first appear 

153 # among the leaves of the tree. 

154 leaves = hierarchy.leaves_list(tree) 

155 order: dict[int, int] = {} 

156 for leaf in leaves: 

157 order.setdefault(int(found[leaf]), len(order) + 1) 

158 numbered = pd.Series( 

159 [order[int(c)] for c in found], index = features.index, 

160 name = 'cluster') 

161 means = features.groupby(numbered).mean() 

162 spread = features.std(ddof = 1).replace(0, np.nan) 

163 standardized = (means - features.mean()) / spread 

164 labels, distance = _name( 

165 standardized, list(names) if names is not None else ( 

166 list(STYLES) if len(means) == len(STYLES) else None)) 

167 profiles = pd.concat([ 

168 pd.DataFrame({ 

169 'style': labels, 'judges': numbered.value_counts().sort_index()}), 

170 means, 

171 standardized.add_suffix('_z'), 

172 pd.DataFrame({'distance': distance})], axis = 1) 

173 profiles.index.name = 'cluster' 

174 gaps = means.to_numpy()[:, None, :] - means.to_numpy()[None, :, :] 

175 counts = (np.abs(gaps) >= spread.to_numpy()).sum(axis = 2) 

176 differences = pd.DataFrame(counts, index = means.index, columns = means.index) 

177 assignments = pd.DataFrame({ 

178 'cluster': numbered, 'style': numbered.map(labels)}) 

179 return Typology( 

180 assignments = assignments, profiles = profiles, 

181 differences = differences)

184def _name( 

185 standardized: pd.DataFrame, 

186 names: list[str] | None) -> tuple[pd.Series, pd.Series]: 

187 """Returns the style of each cluster and its distance from the style. 

188 

189 Args: 

190 standardized: the clusters' mean scores, in standard deviations. 

191 names: the styles to match, which must have profiles in 

192 `PROTOTYPES` and be as many as the clusters, or `None`. 

193 

194 Returns: 

195 The style of each cluster, and the distance between its means and 

196 the style's profile (missing if the clusters were not matched to 

197 styles). 

198 

199 """ 

200 if names is None or len(names) != len(standardized): 

201 labels = pd.Series( 

202 [f'Cluster {c}' for c in standardized.index], 

203 index = standardized.index) 

204 return labels, pd.Series(np.nan, index = standardized.index) 

205 profiles = PROTOTYPES.loc[names, list(_FEATURES)].to_numpy() 

206 means = standardized[list(_FEATURES)].fillna(0).to_numpy() 

207 costs = ((means[:, None, :] - profiles[None, :, :]) ** 2).sum(axis = 2) 

208 rows, columns = optimize.linear_sum_assignment(costs) 

209 labels = pd.Series(index = standardized.index, dtype = object) 

210 distance = pd.Series(index = standardized.index, dtype = 'float64') 

211 for row, column in zip(rows, columns, strict = True): 

212 labels.iloc[row] = names[column] 

213 distance.iloc[row] = float(np.sqrt(costs[row, column])) 

214 return labels, distance
