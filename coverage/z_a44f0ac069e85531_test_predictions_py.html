21def test_add_predictors(simulation: Simulation, scored: pd.DataFrame) -> None: 

22 votes = simulation.votes 

23 added = predictions.add_predictors(votes, scored) 

24 assert added.index.equals(votes.index) 

25 assert set(predictions.PREDICTORS) - {'holdout'} <= set(added.columns) 

26 first = added[added['case_id'] == 'c0'] 

27 ideology = first['judge_ideology'].to_numpy() 

28 # Each judge's colleagues are the other two judges of the case. 

29 for place in range(3): 

30 others = np.delete(ideology, place).mean() 

31 assert first['colleagues_ideology'].iloc[place] == pytest.approx(others) 

32 assert first['panel_ideology_median'].iloc[0] == pytest.approx( 

33 np.median(ideology)) 

34 assert first['panel_ideology_net'].iloc[0] == pytest.approx(ideology.sum()) 

35 assert (added['judge_style'] == 'Collegial').all() 

36 assert added['judge_scored'].all() 

37 # The pressures have the signs that the studies predicted. 

38 opposed = added['district_opposed'].astype(bool) 

39 partisan = added['judge_partisanship'] > 50 

40 assert (added.loc[opposed & partisan, 'partisan_pressure'] > 0).all() 

41 assert (added.loc[~opposed & partisan, 'partisan_pressure'] < 0).all() 

42 deferential = added['review_standard'] == 'deferential' 

43 assert (added.loc[~deferential, 'activist_pressure'] == 0).all() 

44 assert added['expected_reversal'].between(0, 1).all() 

45 assert (added['concurred_only'] & added['judge_dissented']).sum() == 0
